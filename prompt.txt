# 💻 BACKEND (FASTAPI - SIMPLE)

### `app.py`

```python
from fastapi import FastAPI
from model.predict import predict_stress
from data.simulate import generate_data

app = FastAPI()

@app.get("/")
def home():
    return {"message": "Stress Detection API Running"}

@app.get("/predict")
def predict():
    data = generate_data()
    result = predict_stress(data)
    return {"stress_level": result}
```

---

### `simulate.py`

```python
import numpy as np

def generate_data():
    heart_rate = np.random.randint(60, 110)
    gsr = np.random.uniform(0.2, 1.0)
    return [heart_rate, gsr]
```

---

### `predict.py`

```python
def predict_stress(data):
    heart_rate, gsr = data

    if heart_rate > 95:
        return "Severe"
    elif heart_rate > 85:
        return "Moderate"
    elif heart_rate > 75:
        return "Mild"
    else:
        return "No Stress"
```

---

### `Home.jsx`

```javascript
import React, { useState } from "react";
import { getPrediction } from "../services/api";

const Home = () => {
  const [result, setResult] = useState("");

  const handleClick = async () => {
    const data = await getPrediction();
    setResult(data.stress_level);
  };

  return (
    <div>
      <h1>Stress Detection System</h1>
      <button onClick={handleClick}>Predict Stress</button>
      <h2>{result}</h2>
    </div>
  );
};

export default Home;
```
